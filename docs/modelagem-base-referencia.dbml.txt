/* =========================================================
   01. ADMINISTRAÇÃO IZI
   ========================================================= */

Table usuarios_izi {
  id uuid [pk, default: `gen_random_uuid()`]
  nome varchar(100) [not null]
  apelido varchar(100)
  email varchar(150) [not null, unique]
  telefone varchar(30)
  senha varchar(255) [not null]
  foto varchar(500)
  cargo varchar(150)
  status varchar(30) [not null, default: 'ATIVO']
  email_verificado_em timestamp
  ultimo_acesso_em timestamp
  criado_em timestamp [not null, default: `now()`]
  atualizado_em timestamp [not null, default: `now()`]
  eliminado_em timestamp

  Indexes {
    status
    email
  }
}

Table perfis_izi {
  id uuid [pk, default: `gen_random_uuid()`]
  nome varchar(100) [not null]
  descricao text
  status varchar(30) [not null, default: 'ATIVO']
  criado_em timestamp [not null, default: `now()`]
  atualizado_em timestamp [not null, default: `now()`]

  Indexes {
    nome [unique]
    status
  }
}

Table permissoes_izi {
  id uuid [pk, default: `gen_random_uuid()`]
  nome varchar(150) [not null]
  chave varchar(150) [not null, unique]
  modulo varchar(100) [not null]
  descricao text
  criado_em timestamp [not null, default: `now()`]
  atualizado_em timestamp [not null, default: `now()`]

  Indexes {
    modulo
    chave
  }
}

Table usuario_perfil_izi {
  id uuid [pk, default: `gen_random_uuid()`]
  usuario_izi_id uuid [not null]
  perfil_izi_id uuid [not null]
  criado_em timestamp [not null, default: `now()`]

  Indexes {
    (usuario_izi_id, perfil_izi_id) [unique]
  }
}

Table perfil_permissao_izi {
  id uuid [pk, default: `gen_random_uuid()`]
  perfil_izi_id uuid [not null]

  criado_em timestamp [not null, default: `now()`]

  Indexes {

  }
